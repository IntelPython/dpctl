"""Scalar types and dtype descriptors for dpctl kernel arguments.

Every data type known to :mod:`dpctl` is created by :func:`register_dtype`,
including the ones :mod:`dpctl` defines itself. A data type consists of

- static metadata (name, element size, kind, short type code),
- a pair of functions converting between Python objects and the bytes a device
  expects, and
- the identifier of the kernel argument type used to pass such a value to a
  kernel, or no identifier at all, in which case the bytes are passed using the
  ``raw_kernel_arg`` DPC++ extension.

Because that is all a data type is, types that :mod:`numpy` can not describe,
such as :class:`bfloat16`, are ordinary data types here, and so are data types
defined outside of :mod:`dpctl`.
"""

import builtins

import numpy as np

from . import _dtypes_impl

#: a dedicated one.

class _DTypeInfo:
    """Immutable description of a registered data type."""

    __slots__ = (
        "name",
        "itemsize",
        "kind",
        "code",
        "arg_type_id",
        "pack",
        "unpack",
        "numpy_dtype",
        "scalar_type",
        "aliases",
        "available",
        "unavailable_reason",
    )

    def __init__(self, **kwargs):
        for _attr, _value in kwargs.items():
            setattr(self, _attr, _value)

    def __repr__(self):
        return f"<dpctl dtype info for '{self.name}'>"

[docs]class DpctlScalar:
    """Base class for dpctl scalar kernel argument types.

    A scalar holds the bytes a device expects for a value of its data type.
    Instances are created by calling a data type, either its scalar class
    (``dpctl.int32(5)``) or a :class:`dtype` describing it
    (``dpctl.dtype("int32")(5)``).

    Subclasses are created by :func:`register_dtype` and should not be defined
    directly, since only a registered subclass carries a data type.
    """

    #: Description of the data type of instances of this class.
    _dtype_info = None

    #: Value of the ``DPCTLKernelArgType`` enumerator used to pass instances of
    #: this class to a kernel, or ``None`` if instances are passed using the
    #: ``raw_kernel_arg`` extension.
    _arg_type_id = None

    __slots__ = ("_raw",)

    def __init__(self, value):
        info = type(self)._dtype_info
        if info is None:
            raise TypeError(
                f"{type(self).__name__} does not describe a registered dpctl "
                "data type and can not be instantiated"
            )
        if not info.available:
            raise RuntimeError(info.unavailable_reason)
        raw = info.pack(value)
        if not isinstance(raw, (bytes, bytearray, memoryview)):
            raise TypeError(
                f"Data type '{info.name}' packed {type(value).__name__} into "
                f"{type(raw).__name__}, expected bytes"
            )
        raw = bytes(raw)
        if len(raw) != info.itemsize:
            raise ValueError(
                f"Data type '{info.name}' packed {type(value).__name__} into "
                f"{len(raw)} bytes, expected {info.itemsize}"
            )
        self._raw = raw

    @classmethod
    def is_available(cls):
        """Returns whether this data type is supported by this build of
        ``dpctl``.
        """
        info = cls._dtype_info
        return builtins.bool(info is not None and info.available)

    @property
    def value(self):
        """The value of this scalar, as a Python object."""
        return type(self)._dtype_info.unpack(self._raw)

    @property
    def dtype(self):
        """The :class:`dpctl.dtype` of this scalar."""
        return dtype(type(self))

    def __bytes__(self):
        """The bytes a device expects for this value."""
        return self._raw

    def __repr__(self):
        return f"dpctl.{type(self).__name__}({self.value})"




[docs]def register_dtype(
    name,
    *,
    itemsize,
    kind,
    code,
    pack,
    unpack,
    arg_type_id=None,
    numpy_dtype=None,
    aliases=(),
    scalar_type=None,
    doc=None,
    available=True,
    unavailable_reason=None,
):
    """register_dtype(name, *, itemsize, kind, code, pack, unpack, \
arg_type_id=None, numpy_dtype=None, aliases=(), scalar_type=None, doc=None, \
available=True, unavailable_reason=None)

    Adds a data type to the data types :mod:`dpctl` knows, and returns its
    scalar class.

    Once registered, the data type can be looked up by name, by short type
    code, and by its :mod:`numpy` equivalent if it has one, its scalar class
    can be used as a kernel argument, and its short type code is accepted by
    :class:`dpctl.LocalAccessor` and :class:`dpctl.WorkGroupMemory`.

    Args:
        name (str):
            Name of the data type, unique among registered data types.
        itemsize (int):
            Size in bytes of a value of this data type. Every :class:`bytes`
            object ``pack`` returns must be of this length.
        kind (str):
            Single character identifying the family the data type belongs to,
            following :attr:`numpy.dtype.kind` where applicable, e.g. ``'f'``
            for a floating-point type or ``'V'`` for an opaque one.
        code (str):
            Short type code of the data type, unique among registered data
            types, e.g. ``'f4'``. Where a data type has a :mod:`numpy`
            equivalent this should be the :mod:`numpy` type code.
        pack (callable):
            Called with a value and returning the ``itemsize`` bytes a device
            expects for it.
        unpack (callable):
            Called with ``itemsize`` bytes and returning the value they hold.
        arg_type_id (int, optional):
            Value of the ``DPCTLKernelArgType`` enumerator used to pass such a
            value to a kernel. If ``None``, values are passed to kernels using
            the ``raw_kernel_arg`` DPC++ extension, which does not require the
            type to be known to ``dpctl``, but does require that extension to
            be available.
            Default: ``None``.
        numpy_dtype (optional):
            The :mod:`numpy` data type equivalent to this one, if there is one.
            Making it known here is what lets the data type be looked up from
            its :mod:`numpy` equivalent, and lets :mod:`numpy` scalars of it be
            used as kernel arguments.
            Default: ``None``.
        aliases (tuple, optional):
            Additional names the data type can be looked up by.
            Default: ``()``.
        scalar_type (type, optional):
            A subclass of :class:`DpctlScalar` to use as the scalar class of
            this data type. If ``None``, one is created.
            Default: ``None``.
        doc (str, optional):
            Docstring of the created scalar class.
            Default: ``None``.
        available (bool, optional):
            Whether the data type is supported by this build of ``dpctl``. An
            unavailable data type is registered, so that it can be reported as
            unsupported rather than as unknown, but constructing a scalar of it
            or a :class:`dtype` for it raises :exc:`RuntimeError`.
            Default: ``True``.
        unavailable_reason (str, optional):
            Message of that :exc:`RuntimeError`.
            Default: ``None``.

    Returns:
        type:
            The scalar class of the data type, a subclass of
            :class:`DpctlScalar`.

    Raises:
        TypeError:
            If an argument is of an unexpected type.
        ValueError:
            If a name, an alias, or the type code is already registered, or if
            an argument has an invalid value.
    """
    if not isinstance(name, str) or not name:
        raise TypeError("Data type `name` must be a non-empty string")
    if not isinstance(kind, str) or len(kind) != 1:
        raise ValueError("Data type `kind` must be a single character")
    if not isinstance(code, str) or not code:
        raise TypeError("Data type `code` must be a non-empty string")
    if not isinstance(itemsize, int) or itemsize < 1:
        raise ValueError("Data type `itemsize` must be a positive integer")
    if not callable(pack) or not callable(unpack):
        raise TypeError("Data type `pack` and `unpack` must be callable")
    if arg_type_id is not None and not isinstance(arg_type_id, int):
        raise TypeError("Data type `arg_type_id` must be an integer or None")
    if isinstance(aliases, str):
        raise TypeError("Data type `aliases` must be a sequence of strings")

    names = (name,) + tuple(aliases)
    for _name in names:
        if not isinstance(_name, str) or not _name:
            raise TypeError("Data type names must be non-empty strings")
        if _name in _by_name:
            raise ValueError(f"Data type '{_name}' is already registered")
    if code in _by_code:
        raise ValueError(
            f"Type code '{code}' is already used by data type "
            f"'{_by_code[code].name}'"
        )

    np_dtype = None
    if numpy_dtype is not None:
        np_dtype = np.dtype(numpy_dtype)
        if np_dtype.itemsize != itemsize:
            raise ValueError(
                f"NumPy data type {np_dtype} is of {np_dtype.itemsize} bytes, "
                f"while `itemsize` is {itemsize}"
            )
        if np_dtype.num in _by_numpy_num:
            raise ValueError(
                f"NumPy data type {np_dtype} is already used by data type "
                f"'{_by_numpy_num[np_dtype.num].name}'"
            )

    if scalar_type is not None:
        if not (
            isinstance(scalar_type, type)
            and issubclass(scalar_type, DpctlScalar)
            and scalar_type is not DpctlScalar
        ):
            raise TypeError(
                "Data type `scalar_type` must be a subclass of DpctlScalar"
            )
        if scalar_type in _by_scalar_type:
            raise ValueError(
                f"Class {scalar_type.__name__} is already the scalar class of "
                f"data type '{_by_scalar_type[scalar_type].name}'"
            )
        if scalar_type._dtype_info is not None:
            raise ValueError(
                f"Class {scalar_type.__name__} already carries a data type"
            )

    info = _DTypeInfo(
        name=name,
        itemsize=itemsize,
        kind=kind,
        code=code,
        arg_type_id=arg_type_id,
        pack=pack,
        unpack=unpack,
        numpy_dtype=np_dtype,
        scalar_type=None,
        aliases=tuple(aliases),
        available=builtins.bool(available),
        unavailable_reason=(
            unavailable_reason
            or f"Data type '{name}' is not supported by this build of dpctl"
        ),
    )

    if scalar_type is None:
        scalar_type = type(
            name,
            (DpctlScalar,),
            {
                "__slots__": (),
                "__doc__": doc or f"Scalar of dpctl data type '{name}'.",
                "__module__": "dpctl",
            },
        )
    info.scalar_type = scalar_type
    scalar_type._dtype_info = info
    scalar_type._arg_type_id = arg_type_id

    for _name in names:
        _by_name[_name] = info
    _by_code[code] = info
    _by_scalar_type[scalar_type] = info
    if np_dtype is not None:
        _by_numpy_num[np_dtype.num] = info

    return scalar_type



def _unregister_dtype(name):
    """Removes a data type registered by :func:`register_dtype`.

    Existing :class:`dtype` instances and scalars of the data type keep
    working, but the data type can no longer be looked up.
    """
    info = _by_name.get(name)
    if info is None:
        raise LookupError(f"Data type '{name}' is not registered")
    for _name in (info.name,) + info.aliases:
        _by_name.pop(_name, None)
    _by_code.pop(info.code, None)
    _by_scalar_type.pop(info.scalar_type, None)
    if info.numpy_dtype is not None:
        _by_numpy_num.pop(info.numpy_dtype.num, None)

    """registered_dtypes()

    Returns the data types registered with :func:`register_dtype` and supported
    by this build of ``dpctl``.

    Returns:
        tuple:
            A tuple of :class:`dpctl.dtype` instances.
    """
    seen = {}

[docs]def is_dtype_available(arg):
    """is_dtype_available(arg)

    Returns whether ``arg`` describes a data type that is registered and
    supported by this build of ``dpctl``.

    Args:
        arg:
            Anything :class:`dpctl.dtype` accepts.

    Returns:
        bool:
            ``True`` if a :class:`dpctl.dtype` can be created from ``arg``,
            ``False`` otherwise.
    """
    try:
        dtype(arg)
    except (TypeError, RuntimeError):
        return False
    return True



def _check_available(info):
    if not info.available:
        raise RuntimeError(info.unavailable_reason)
    return info


def _info_from_numpy(arg):
    """Resolves a NumPy data type, or anything NumPy accepts, to the data type
    it is equivalent to.
    """
    if isinstance(arg, np.dtype):
        np_dt = arg
    else:
        try:
            np_dt = np.dtype(arg)
        except (TypeError, ValueError):
            raise TypeError(f"Cannot create a dpctl.dtype from {arg!r}")
    if np_dt.byteorder in ("<", ">"):
        raise TypeError("Only native byte order is supported.")
    info = _by_numpy_num.get(np_dt.num)
    if info is None:
        raise TypeError(f"Cannot create a dpctl.dtype from {arg!r}")
    return _check_available(info)


def _info_from_code(code):
    """Resolves a short type code, or a data type name, to a data type,
    returning ``None`` if it is not registered.
    """
    info = _by_code.get(code)
    if info is None:
        info = _by_name.get(code)
    return info


def _info_from_numpy_num(num):
    """Resolves the ``num`` of a NumPy data type to the data type it is
    equivalent to, returning ``None`` if there is none.
    """
    return _by_numpy_num.get(num)


def _resolve(arg):
    if isinstance(arg, dtype):
        return arg._info
    if isinstance(arg, DpctlScalar):
        arg = type(arg)
    if isinstance(arg, type) and issubclass(arg, DpctlScalar):
        info = _by_scalar_type.get(arg)
        if info is None:
            raise TypeError(f"Cannot create a dpctl.dtype from {arg!r}")
        return _check_available(info)
    if isinstance(arg, str):
        info = _by_name.get(arg)
        if info is None:
            info = _by_code.get(arg)
        if info is not None:
            return _check_available(info)
    return _info_from_numpy(arg)



[docs]class dtype:
    """
    dtype(arg)

    Dpctl dtype descriptor class.

    Analogous to :class:`numpy.dtype`, a :class:`dpctl.dtype` identifies a
    data type and carries static metadata.

    A ``dpctl.dtype`` can be constructed from:

    - A dpctl scalar class, or a scalar of one
    - A data type name, e.g. ``"float32"``, or short type code, e.g. ``"f4"``
    - A NumPy dtype or type equivalent to a registered data type
    - Another :class:`dpctl.dtype`

    Non-native byte orders are rejected.

    Args:
        arg:
            Object to create the data type descriptor from.

    Raises:
        TypeError:
            If ``arg`` does not describe a registered data type.
        RuntimeError:
            If ``arg`` describes a data type that this build of ``dpctl`` does
            not support, such as :class:`dpctl.bfloat16` when the DPC++
            compiler used to build ``dpctl`` did not provide it.
    """

    __slots__ = ("_info",)

    def __init__(self, arg):
        self._info = _resolve(arg)

    @property
    def name(self):
        """The short name of this dtype"""
        return self._info.name

    @property
    def itemsize(self):
        """Element size in bytes."""
        return self._info.itemsize

    @property
    def kind(self):
        """Single-character kind code"""
        return self._info.kind

    @property
    def str(self):
        """Short dtype string"""
        return self._info.code

    @property
    def scalar_type(self):
        """The dpctl scalar class for this dtype"""
        return self._info.scalar_type

    @property
    def numpy_dtype(self):
        """The equivalent :class:`numpy.dtype`, or ``None`` if this data type
        has no NumPy equivalent.
        """
        return self._info.numpy_dtype

    @property
    def arg_type_id(self):
        """The internal ``DPCTLKernelArgType`` enum value, or ``None`` if
        values of this dtype are passed to kernels as raw kernel arguments.
        """
        return self._info.arg_type_id

    @property
    def uses_raw_kernel_arg(self):
        """Whether values of this dtype are passed to kernels using the
        ``raw_kernel_arg`` DPC++ extension.
        """
        return self._info.arg_type_id is None

    def __repr__(self):
        return f"dpctl.dtype('{self._info.name}')"

    def __eq__(self, other):
        if not isinstance(other, dtype):
            try:
                other = dtype(other)
            except (TypeError, RuntimeError):
                return NotImplemented
        return self._info.name == other._info.name

    def __ne__(self, other):
        result = self.__eq__(other)
        if result is NotImplemented:
            return result
        return not result

    def __hash__(self):
        return hash(self._info.name)

    def __call__(self, value):
        """Construct a scalar of this dtype"""
        return self._info.scalar_type(value)



def _numpy_pack(np_type):
    def pack(value):
        return np_type(value).tobytes()

    return pack


def _numpy_unpack(np_dtype):
    def unpack(raw):
        return np.frombuffer(raw, dtype=np_dtype, count=1)[0]

    return unpack


def _pack_bool(value):
    return b"\x01" if builtins.bool(value) else b"\x00"


def _unpack_bool(raw):
    return np.bool_(raw[0] != 0)
